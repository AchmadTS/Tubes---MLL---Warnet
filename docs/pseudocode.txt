type Infotype_Billing <
    idBilling   : string
    namaUser    : string
    jamMulai    : string
    durasi      : integer
    totalBayar  : integer
>

type Adr_Billing : pointer to Elm_Billing
type Elm_Billing <
    info : Infotype_Billing
    next : Adr_Billing
>

type Infotype_Komputer <
    idPC        : string
    namaPC      : string
    tipe        : string
    tarifPerJam : integer
    status      : string
>

type Adr_Komputer : pointer to Elm_Komputer
type Elm_Komputer <
    info        : Infotype_Komputer
    nextBilling : Adr_Billing
    next        : Adr_Komputer
>

type List_Warnet <
    first : Adr_Komputer
>

// --- INISIALISASI & OPERASI PARENT (KOMPUTER) ---

Procedure createList(In/Out L : List_Warnet)
{
    I.S. : Terdefinisi List_Warnet L sembarang
    F.S. : Terbentuk List_Warnet L kosong (L.first = NIL)
}
Kamus
    -
Algoritma
    L.first = NIL
Endprocedure

Function createKomputer(idPC, namaPC, tipe : string, tarifPerJam : integer, status : string) -> Adr_Komputer
{
    I.S. : Terdefinisi idPC, namaPC, tipe, tarifPerJam, dan status
    F.S. : Mengembalikan alamat elemen komputer baru yang sudah dialokasi dengan info sesuai parameter, serta nextBilling = NIL dan next = NIL
}
Kamus
    K : Adr_Komputer
Algoritma
    allocate(K)
    K->info.idPC        = idPC
    K->info.namaPC      = namaPC
    K->info.tipe        = tipe
    K->info.tarifPerJam = tarifPerJam
    K->info.status      = status
    K->nextBilling      = NIL
    K->next             = NIL
    return K
Endfunction

Function findKomputer(L : List_Warnet, idPC : string) -> Adr_Komputer
{
    I.S. : Terdefinisi List_Warnet L yang mungkin kosong, dan idPC
    F.S. : Mengembalikan alamat elemen komputer dengan idPC yang sesuai pada L, atau NIL jika tidak ditemukan
}
Kamus
    K : Adr_Komputer
Algoritma
    K = L.first
    while K != NIL and K->info.idPC != idPC do
        K = K->next
    endwhile
    return K
Endfunction

Procedure insertKomputer(In/Out L : List_Warnet, In K : Adr_Komputer)
{
    I.S. : Terdefinisi List_Warnet L yang mungkin kosong, dan K sudah dialokasi
    F.S. : K ditambahkan sebagai elemen terakhir pada L dan tampilkan pesan berhasil jika idPC belum terdaftar, jika sudah terdaftar, tampilkan pesan "ID Komputer sudah terdaftar"
}
Kamus
    Q : Adr_Komputer
Algoritma
    if findKomputer(L, K->info.idPC) != NIL then
        output("ID Komputer sudah terdaftar")
    else
        if L.first == NIL then
            L.first = K
        else
            Q = L.first
            while Q->next != NIL do
                Q = Q->next
            endwhile
            Q->next = K
        endif
        output("Komputer [", K->info.idPC, "] berhasil ditambahkan!")
    endif
Endprocedure

Procedure deleteKomputer(In/Out L : List_Warnet, In idPC : string)
{
    I.S. : Terdefinisi List_Warnet L yang mungkin kosong, dan idPC
    F.S. : Jika idPC ditemukan, seluruh node billing pada komputer tersebut beserta node komputernya dihapus, didealokasi, dan tampilkan pesan berhasil, jika tidak ditemukan, tampilkan pesan "Komputer tidak ditemukan!"
}
Kamus
    K, prec : Adr_Komputer
    B       : Adr_Billing
Algoritma
    K = findKomputer(L, idPC)
    if K == NIL then
        output("Komputer tidak ditemukan!")
    else
        while K->nextBilling != NIL do
            B = K->nextBilling
            K->nextBilling = B->next
            B->next = NIL
            deallocate(B)
        endwhile

        if K == L.first then
            L.first = K->next
        else
            prec = L.first
            while prec->next != K do
                prec = prec->next
            endwhile
            prec->next = K->next
        endif
        K->next = NIL
        deallocate(K)
        output("Komputer [", idPC, "] beserta seluruh riwayat billing-nya berhasil dihapus!")
    endif
Endprocedure

Procedure printKomputer(In L : List_Warnet)
{
    I.S. : Terdefinisi List_Warnet L yang mungkin kosong
    F.S. : Seluruh data komputer pada L ditampilkan ke layar, atau tampilkan pesan "Data komputer kosong" jika L kosong
}
Kamus
    K : Adr_Komputer
Algoritma
    if L.first == NIL then
        output("Data komputer kosong")
    else
        K = L.first
        while K != NIL do
            output(K->info.idPC, K->info.namaPC, K->info.tipe, K->info.tarifPerJam, K->info.status)
            K = K->next
        endwhile
    endif
Endprocedure


// --- OPERASI CHILD (BILLING) ---

Function createBilling(idBilling, namaUser, jamMulai : string, durasi : integer) -> Adr_Billing
{
    I.S. : Terdefinisi idBilling, namaUser, jamMulai, dan durasi
    F.S. : Mengembalikan alamat elemen billing baru yang telah dialokasi dengan info sesuai parameter, totalBayar = 0, dan next = NIL
}
Kamus
    B : Adr_Billing
Algoritma
    allocate(B)
    B->info.idBilling  = idBilling
    B->info.namaUser   = namaUser
    B->info.jamMulai   = jamMulai
    B->info.durasi     = durasi
    B->info.totalBayar = 0
    B->next            = NIL
    return B
Endfunction

Function findBilling(K : Adr_Komputer, idBilling : string) -> Adr_Billing
{
    I.S. : Terdefinisi K yang mungkin NIL, dan idBilling
    F.S. : Mengembalikan alamat elemen billing dengan idBilling yang sesuai pada komputer K, atau NIL jika K = NIL atau data tidak ditemukan
}
Kamus
    B : Adr_Billing
Algoritma
    if K == NIL then
        return NIL
    else
        B = K->nextBilling
        while B != NIL and B->info.idBilling != idBilling do
            B = B->next
        endwhile
        return B
    endif
Endfunction

Procedure insertBilling(In/Out K : Adr_Komputer, In B : Adr_Billing)
{
    I.S. : Terdefinisi K yang mungkin NIL, dan B sudah dialokasi
    F.S. : Jika K != NIL dan idBilling belum terdaftar, totalBayar B dihitung (durasi * tarifPerJam), B ditambahkan sebagai elemen terakhir pada list billing K, dan tampilkan pesan berhasil beserta totalBayar, jika tidak valid, tampilkan pesan error
}
Kamus
    Q : Adr_Billing
Algoritma
    if K == NIL then
        output("Komputer tidak ditemukan!")
    else if findBilling(K, B->info.idBilling) != NIL then
        output("ID Billing sudah terdaftar pada komputer ini!")
    else
        B->info.totalBayar = B->info.durasi * K->info.tarifPerJam
        if K->nextBilling == NIL then
            K->nextBilling = B
        else
            Q = K->nextBilling
            while Q->next != NIL do
                Q = Q->next
            endwhile
            Q->next = B
        endif
        output("Billing [", B->info.idBilling, "] ditambahkan ke [", K->info.idPC, "] | Total Bayar: Rp", B->info.totalBayar)
    endif
Endprocedure

Procedure deleteBilling(In/Out K : Adr_Komputer, In idBilling : string)
{
    I.S. : Terdefinisi K yang mungkin NIL, dan idBilling
    F.S. : Jika K != NIL dan idBilling ditemukan, node billing tersebut dihapus dari list billing K, didealokasi, dan tampilkan pesan berhasil, jika tidak ditemukan, tampilkan pesan error
}
Kamus
    B, prec : Adr_Billing
Algoritma
    if K == NIL then
        output("Komputer tidak ditemukan!")
    else
        B = findBilling(K, idBilling)
        if B == NIL then
            output("Data billing tidak ditemukan!")
        else
            if B == K->nextBilling then
                K->nextBilling = B->next
            else
                prec = K->nextBilling
                while prec->next != B do
                    prec = prec->next
                endwhile
                prec->next = B->next
            endif
            B->next = NIL
            deallocate(B)
            output("Billing [", idBilling, "] pada [", K->info.idPC, "] berhasil dihapus!")
        endif
    endif
Endprocedure

Procedure printBilling(In K : Adr_Komputer)
{
    I.S. : Terdefinisi K yang mungkin NIL
    F.S. : Seluruh riwayat billing pada komputer K ditampilkan ke layar, atau tampilkan pesan jika komputer tidak ditemukan atau riwayat kosong
}
Kamus
    B : Adr_Billing
Algoritma
    if K == NIL then
        output("Komputer tidak ditemukan")
    else if K->nextBilling == NIL then
        output("Belum ada riwayat billing pada komputer ini")
    else
        B = K->nextBilling
        while B != NIL do
            output(B->info.idBilling, B->info.namaUser, B->info.jamMulai, B->info.durasi, B->info.totalBayar)
            B = B->next
        endwhile
    endif
Endprocedure

Procedure printAll(In L : List_Warnet)
{
    I.S. : Terdefinisi List_Warnet L yang mungkin kosong
    F.S. : Seluruh data komputer beserta masing-masing riwayat billing-nya ditampilkan ke layar, atau tampilkan pesan "Data warnet kosong" jika L kosong
}
Kamus
    K : Adr_Komputer
Algoritma
    if L.first == NIL then
        output("Data warnet kosong")
    else
        K = L.first
        while K != NIL do
            output(K->info.idPC, K->info.namaPC, K->info.tipe, K->info.tarifPerJam, K->info.status)
            printBilling(K)
            K = K->next
        endwhile
    endif
Endprocedure

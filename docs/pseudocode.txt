type Infotype_Billing <
    idBilling   : string
    namaUser    : string
    jamMulai    : string
    durasi      : integer
    totalBayar  : integer
>

type Adr_Billing : pointer to Elm_Billing
type Elm_Billing <
    info : Infotype_Billing
    next : Adr_Billing
>

type Infotype_Komputer <
    idPC        : string
    namaPC      : string
    tipe        : string
    tarifPerJam : integer
    status      : string
>

type Adr_Komputer : pointer to Elm_Komputer
type Elm_Komputer <
    info        : Infotype_Komputer
    nextBilling : Adr_Billing
    next        : Adr_Komputer
>

type List_Warnet <
    first : Adr_Komputer
>


// =========================================================
// OPERASI WAJIB MLL (1-TO-N)
// =========================================================

// --- INISIALISASI & OPERASI PARENT (KOMPUTER) ---

Procedure createList(In/Out L : List_Warnet)
Kamus
    -
Algoritma
    L.first = NIL
Endprocedure

Function createKomputer(idPC, namaPC, tipe : string, tarifPerJam : integer, status : string) -> Adr_Komputer
Kamus
    K : Adr_Komputer
Algoritma
    allocate(K)
    K->info.idPC        = idPC
    K->info.namaPC      = namaPC
    K->info.tipe        = tipe
    K->info.tarifPerJam = tarifPerJam
    K->info.status      = status
    K->nextBilling      = NIL
    K->next             = NIL
    return K
Endfunction

Function findKomputer(L : List_Warnet, idPC : string) -> Adr_Komputer
Kamus
    K : Adr_Komputer
Algoritma
    K = L.first
    while K != NIL and K->info.idPC != idPC do
        K = K->next
    endwhile
    return K
Endfunction

Procedure insertKomputer(In/Out L : List_Warnet, In K : Adr_Komputer)
Kamus
    Q : Adr_Komputer
Algoritma
    if findKomputer(L, K->info.idPC) != NIL then
        output("ID Komputer sudah terdaftar")
    else
        if L.first == NIL then
            L.first = K
        else
            Q = L.first
            while Q->next != NIL do
                Q = Q->next
            endwhile
            Q->next = K
        endif
    endif
Endprocedure

Procedure deleteKomputer(In/Out L : List_Warnet, in idPC : string)
Kamus
    K, prec : Adr_Komputer
    B       : Adr_Billing
Algoritma
    K = findKomputer(L, idPC)
    if K == NIL then
        output("Komputer tidak ditemukan")
    else
        while K->nextBilling != NIL do
            B = K->nextBilling
            K->nextBilling = B->next
            B->next = NIL
            deallocate(B)
        endwhile

        if K == L.first then
            L.first = K->next
        else
            prec = L.first
            while prec->next != K do
                prec = prec->next
            endwhile
            prec->next = K->next
        endif
        K->next = NIL
        deallocate(K)
    endif
Endprocedure

Procedure printKomputer(L : List_Warnet)
Kamus
    K : Adr_Komputer
Algoritma
    if L.first == NIL then
        output("Data komputer kosong")
    else
        K = L.first
        while K != NIL do
            output(K->info.idPC, K->info.namaPC, K->info.tipe, K->info.tarifPerJam, K->info.status)
            K = K->next
        endwhile
    endif
Endprocedure
